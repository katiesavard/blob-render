103:	
104:	#create a -model fits file with the simulated data in it with the same format as the -image file produced from the previous cleaning step 
105:	f.write(
106:    	f"python3 -m blobrender.tools.populatefits --model_fits {fitsfile_name} "
107:    	f"--wsclean_fits {predict_file_name}-image.fits "
108:    	f"--op_fits {predict_file_name}-model.fits "
109:		f"--nchan {nchannels}\n"
110:	)
111:	
112:	#change the RA and DEC of the model fits files to the desired position

124:	f.write('pixscale=$(python3 -m blobrender.tools.calc_beamsize --fitsfile '+predict_file_name+' --pixels_per_beam '+pixels_per_beam+' --nchan '+nchannels+')\n')
